%% Title block
\NewDocumentCommand{\makepapertitle}{}{%
  \pagenumbering{arabic}%
  \chapter*{\PaperTitle}%
  \markboth{\ShortTitle}{\ShortTitle}%
  \vspace*{-12pt}%
  {\setlength{\parskip}{0pt}%
   {\sffamily\large\color{Gray40}\PaperSubtitle\par}%
   \ifdefined\PaperCredits
     {\sffamily\small\color{Gray40}\PaperCredits\par}%
   \fi
   \ifdefined\PaperAcknowledgment
     \vspace{\baselineskip}{\sffamily\small\color{Gray40}\PaperAcknowledgment\par}%
   \fi}%
  \vspace{14pt}%
}
